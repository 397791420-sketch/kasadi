1. 打开 supabase.com 注册免费账号，创建项目
2. 左侧 SQL Editor 执行下面SQL建表（复制粘贴整段）：
create table if not exists reports (id bigint primary key, data text, updated_at timestamptz default now());
create table if not exists customers (id bigint primary key, data text, updated_at timestamptz default now());
alter table reports disable row level security;
alter table customers disable row level security;
